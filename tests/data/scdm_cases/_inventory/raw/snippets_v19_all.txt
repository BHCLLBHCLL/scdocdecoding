numLayers = 10

def CreateBallInComponent(radius):
	ballComp = ComponentHelper.CreateAtRoot("Ball")
	ComponentHelper.SetActive(ballComp)
	s = Sphere.Create(Frame.World, MM(r))
	result = SurfaceBody.Create(s, BoxUV.Empty)
	ComponentHelper.SetRootActive()
	return ballComp

def CopyComponent(comp, x, y, z):
	matrix = Matrix.CreateTranslation(Vector.Create(MM(x),MM(y),MM(z)))
	comp = ComponentHelper.CopyToRoot(comp)
	comp.Transform(matrix)

#create a ball in a component for copying
ballComp = CreateBallInComponent(r)

# run nested while loops to create balls in x and y and all layers
layer = 0
while layer < numLayers:
	x = 0
	while x <= layer:
		y = 0
		while y <= layer:
			if layer%2 == 0:
				CopyComponent(ballComp, 2*r*x -r*layer, 2*r*y -r*layer, -1.414*r*layer)
			if layer%2 == 1:
				CopyComponent(ballComp, 2*r*x  -r*layer, 2*r*y  -r*layer, -1.414*r*layer)
			y = y+1
		x = x + 1
	layer = layer + 1
  

# delete the template ball
ballComp.Delete()

import math

outerRadius = 20
innerRadius = 18
outsideCircle = Circle.Create(Frame.World, MM(outerRadius))
insideCircle = Circle.Create(Frame.World, MM(innerRadius))

boundary = List[ITrimmedCurve]();
inwardLine = Line.Create(Point.Origin, -Direction.DirX);
outwardLine = Line.Create(Point.Origin, Direction.DirX);
axis = outsideCircle.Axis;
nTeeth = 20
repeatAngle = 2 * math.pi / nTeeth
toothAngle = 0.6 * repeatAngle
gapAngle = repeatAngle - toothAngle

i = 0
while i < nTeeth:
	startTooth = i * repeatAngle;
	endTooth = startTooth + toothAngle;
	boundary.Add(CurveSegment.Create(outsideCircle, Interval.Create(startTooth, endTooth)))

	# rotate 'inwardLine' about the circle axis
	rotatedInwardLine = Matrix.CreateRotation(axis, endTooth) * inwardLine
	
	# a line segment is just a parameter interval of an unbounded line
	boundary.Add(CurveSegment.Create(rotatedInwardLine, Interval.Create(MM(-outerRadius), MM(-innerRadius))))

	startGap = endTooth
	endGap = startGap + gapAngle
	boundary.Add(CurveSegment.Create(insideCircle, Interval.Create(startGap, endGap)))

	rotatedOutwardLine = Matrix.CreateRotation(axis, endGap) * outwardLine
	boundary.Add(CurveSegment.Create(rotatedOutwardLine, Interval.Create(MM(innerRadius), MM(outerRadius))))
	i = i + 1
	
hole = Circle.Create(Frame.Create(Point.Origin, Direction.DirX, Direction.DirY), MM(5))
boundary.Add(CurveSegment.Create(hole))

designResult = PlanarBody.Create(Plane.PlaneXY, boundary)
designBody = designResult.CreatedBody

selection = Selection.Create(designBody.Faces[0])
options = ExtrudeFaceOptions()

        DatumPoint.Create(GetRootPart(), "pt", point)

        DatumLine.Create(GetRootPart(), "line", line)

def SplitRounds(desBody, maxRadius):
    if not type(desBody) is DesignBody:
        raise Exception("Could not get a design body") 
    
    rounds = List[IDesignFace]()
    curves = List[ITrimmedCurve]()
    for desFace in desBody.Faces:
        roundInfo = RoundInfo.Create(desFace)
        if roundInfo.Success:
            if roundInfo.Radius <= maxRadius:
                rounds.Add(desFace)
                for curve in desFace.GetIsoCurves(roundInfo.AlongU, 0.5):
                    if curve is not None:
                        curves.Add(curve)

    if not any(rounds):
        raise Exception("No rounds found")
        
    SplitFace.ByCurves(Selection.Create(rounds), curves)

groups = WindowHelper.GetGroups()
